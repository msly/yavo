use std/io

struct Version {
    original: String
    major: int
    minor: int
    patch: int
    pre: Vec[String]
}

fn coreNumber(text: str) -> ?int {
    if text.len() == 0 { return none }
    var value = 0
    for i in 0..text.len() {
        if text[i] < '0' || text[i] > '9' { return none }
        value = value * 10 + int(text[i]) - int('0')
        if value > 1000000000 { return none }
    }
    return value
}

fn validIdentifiers(text: str) -> bool {
    var length = 0
    for i in 0..text.len() {
        let c = text[i]
        if c == '.' {
            if length == 0 { return false }
            length = 0
        } else {
            if !((c >= 'a' && c <= 'z') || (c >= 'A' && c <= 'Z') || (c >= '0' && c <= '9') || c == '-') { return false }
            length += 1
            if length > 32 { return false }
        }
    }
    return length > 0
}

fn parse(text: str) -> ?Version {
    let plus = text.find("+") ?? text.len()
    if plus < text.len() {
        if !validIdentifiers(text[plus + 1..text.len()]) { return none }
    }
    let prefix = text[0..plus]
    let dash = prefix.find("-") ?? prefix.len()
    let core = prefix[0..dash].split(".")
    if core.len() != 3 { return none }
    let major = coreNumber(core[0]) ?? return none
    let minor = coreNumber(core[1]) ?? return none
    let patch = coreNumber(core[2]) ?? return none
    var pre: Vec[String] = []
    if dash < prefix.len() {
        let identifiers = prefix[dash + 1..prefix.len()]
        if !validIdentifiers(identifiers) { return none }
        for identifier in identifiers.split(".") { pre.push(identifier.toString()) }
    }
    return Version{original: text.toString(), major: major, minor: minor, patch: patch, pre: pre}
}

fn numeric(text: str) -> bool {
    for i in 0..text.len() {
        if text[i] < '0' || text[i] > '9' { return false }
    }
    return true
}

fn compareIdentifier(a: str, b: str) -> int {
    let an = numeric(a)
    let bn = numeric(b)
    if an && !bn { return -1 }
    if !an && bn { return 1 }
    if an && bn {
        var ai = 0
        var bi = 0
        while ai + 1 < a.len() && a[ai] == '0' { ai += 1 }
        while bi + 1 < b.len() && b[bi] == '0' { bi += 1 }
        let av = a[ai..a.len()]
        let bv = b[bi..b.len()]
        if av.len() < bv.len() { return -1 }
        if av.len() > bv.len() { return 1 }
        if av < bv { return -1 }
        if av > bv { return 1 }
        return 0
    }
    if a < b { return -1 }
    if a > b { return 1 }
    return 0
}

fn before(a: Version, b: Version) -> bool {
    if a.major != b.major { return a.major < b.major }
    if a.minor != b.minor { return a.minor < b.minor }
    if a.patch != b.patch { return a.patch < b.patch }
    if a.pre.len() == 0 { return false }
    if b.pre.len() == 0 { return true }
    for i in 0..min(a.pre.len(), b.pre.len()) {
        let comparison = compareIdentifier(a.pre[i], b.pre[i])
        if comparison != 0 { return comparison < 0 }
    }
    return a.pre.len() < b.pre.len()
}

fn main() {
    let text = io.readAll() catch _ { return }
    var versions: Vec[Version] = []
    var invalid = 0
    for index, line in text.lines() {
        if let version = parse(line) {
            versions.push(version)
        } else {
            println("line ${index + 1}: invalid")
            invalid += 1
        }
    }
    versions.sortBy((a, b) => before(a, b))
    for version in versions { println(version.original) }
    println("valid: ${versions.len()} invalid: ${invalid}")
}
