use std/io

fn matches(pattern: str, text: str) -> String {
    var tokens: Vec[int] = []
    var pos = 0
    while pos < pattern.len() {
        let current = pattern[pos]
        if current == '\\' {
            pos += 1
            if pos == pattern.len() { return "invalid".toString() }
            tokens.push(int(pattern[pos]))
        } else if current == '*' {
            tokens.push(-1)
        } else if current == '?' {
            tokens.push(-2)
        } else {
            tokens.push(int(current))
        }
        pos += 1
    }
    var previous: Vec[bool] = []
    for i in 0..text.len() + 1 { previous.push(i == 0) }
    for token in tokens {
        var next: Vec[bool] = []
        for _i in 0..text.len() + 1 { next.push(false) }
        if token == -1 {
            next[0] = previous[0]
            for j in 1..text.len() + 1 { next[j] = previous[j] || next[j - 1] }
        } else {
            for j in 1..text.len() + 1 {
                next[j] = previous[j - 1] && (token == -2 || token == int(text[j - 1]))
            }
        }
        previous = next
    }
    if previous[text.len()] { return "yes".toString() }
    return "no".toString()
}

fn main() {
    let text = io.readAll() catch _ { return }
    let lines = text.lines()
    for group in 0..lines.len() / 2 {
        println("${group + 1}: ${matches(lines[group * 2], lines[group * 2 + 1])}")
    }
}
