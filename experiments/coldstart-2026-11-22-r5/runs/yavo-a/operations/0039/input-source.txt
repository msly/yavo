use std/io

fn show(cache: Map[String, (int, int, int)]) {
    if cache.len() == 0 {
        println("keys: none")
        return
    }
    var ordered: Vec[(String, int)] = []
    for key, entry in cache {
        ordered.push((key.clone(), entry.2))
    }
    ordered.sortBy((a, b) => a.1 > b.1)
    var names: Vec[String] = []
    for entry in ordered {
        names.push(entry.0.clone())
    }
    println("keys: ${names.join(" ")}")
}

fn main() {
    let input = io.readAll() catch _ { return }
    let lines = input.lines()
    let capacity = lines[0].trim().parseInt() ?? 0
    var cache: Map[String, (int, int, int)] = [:]
    for row in 1..lines.len() {
        let fields = lines[row].fields()
        if fields.len() == 0 {
            continue
        }
        let time = fields[0].parseInt() ?? 0
        var expired: Vec[String] = []
        for key, entry in cache {
            if entry.1 <= time {
                expired.push(key.clone())
            }
        }
        for key in expired {
            let _ = cache.remove(key)
        }
        if fields[1] == "put" {
            let key = fields[2]
            let value = fields[3].parseInt() ?? 0
            let ttl = fields[4].parseInt() ?? 0
            if ttl == 0 || capacity == 0 {
                let _ = cache.remove(key)
            } else {
                cache.set(key.toString(), (value, time + ttl, row))
                if cache.len() > capacity {
                    var oldest = row + 1
                    var victim = String{}
                    for candidate, entry in cache {
                        if entry.2 < oldest {
                            oldest = entry.2
                            victim = candidate.clone()
                        }
                    }
                    let _ = cache.remove(victim)
                }
            }
        } else if fields[1] == "get" {
            let key = fields[2]
            if let entry = cache.getCopy(key) {
                println("${key}: ${entry.0}")
                cache.set(key.toString(), (entry.0, entry.1, row))
            } else {
                println("${key}: missing")
            }
        } else {
            show(cache)
        }
    }
    show(cache)
}
